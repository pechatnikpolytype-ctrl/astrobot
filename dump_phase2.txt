async def _process_referral(new_user_id: int, referrer_id: int):
    if new_user_id == referrer_id:
        return
    settings = await get_referral_settings()
    if not settings.get("enabled", True):
        return
    max_bonuses = settings.get("max_bonuses", 10)
    if max_bonuses > 0:
        count = await count_user_referrals_with_bonus(referrer_id)
        if count >= max_bonuses:
            return
    bonus_days = settings.get("bonus_days", 3)
    added = await add_referral(referrer_id, new_user_id, bonus_days)
    if not added:
        return
    await set_subscription(new_user_id, bonus_days)
    await set_subscription(referrer_id, bonus_days)
    await mark_referral_bonus_paid(new_user_id)
    try:
        from loader import bot
        await bot.send_message(
            referrer_id,
            f"🎉 По вашей ссылке зарегистрировался новый пользователь!\n\n"
            f"Вам начислено {bonus_days} дней Премиум."
        )
    except Exception:
        pass


@router.message(Command("start"))
async def start_handler(message: Message, command: CommandObject = None):
    user_id = message.from_user.id
    username = message.from_user.username or ""
    first_name = message.from_user.first_name or ""

    if command and command.args and command.args.startswith("ref_"):
        try:
            referrer_id = int(command.args.replace("ref_", ""))
            existing = await get_user(user_id)
            if not existing:
                await create_user(user_id, username, first_name)
                await _process_referral(user_id, referrer_id)
        except Exception as e:
            logger.warning(f"Ошибка реферальной ссылки: {e}")

    await create_user(user_id, username, first_name)

    # Отправляем меню с фактом через menu_commands
    from handlers.menu_commands import show_main_menu_from_start
    await show_main_menu_from_start(message, bot=None)

@router.message(NatalChartStates.waiting_birth_place)
async def natal_finish(message: Message, state: FSMContext):
    if message.text == "/cancel":
        await state.clear()
        await message.answer("Отменено", reply_markup=back_to_menu_keyboard())
        return

    place = message.text.strip()
    valid, cleaned, error = natal_service.validate_place(place)
    if not valid:
        await message.answer(f"❌ {error}\n\nПопробуйте снова или /cancel")
        return

    data = await state.get_data()
    data["birth_place"] = cleaned
    await state.clear()

    processing_msg = await message.answer(
        "🔮 Создаю натальную карту...\n\n"
        "Это может занять 20-30 секунд."
    )

    try:
        # 1. Создаём карту (VedAstro)
        chart_data = await natal_service.create_natal_chart(
            data["name"], data["birth_date"], data["birth_time"], data["birth_place"]
        )
        user_id = message.from_user.id
        last_chart_cache[user_id] = chart_data

        await update_birth_data(
            user_id, data["birth_date"], data["birth_time"], data["birth_place"]
        )

        try:
            await processing_msg.delete()
        except Exception:
            pass

        # 2. AI-интерпретация (с кэшем)
        ai_processing = await message.answer("🔮 Астролог анализирует вашу карту...")
        try:
            ai_text = await natal_service.generate_ai_report(chart_data)
            if ai_text:
                # Ограничение на PDF
                chart_data["ai_interpretation"] = ai_text[:1500]
                logger.info(f"✅ AI-интерпретация: {len(ai_text)} символов")
            else:
                chart_data["ai_interpretation"] = ""
                logger.warning("AI не дал интерпретацию")
        except Exception as e:
            logger.error(f"Ошибка AI: {e}")
            chart_data["ai_interpretation"] = ""

        try:
            await ai_processing.delete()
        except Exception:
            pass

        # 3. Отправляем SVG-схему
        image_path = natal_service.generate_svg_chart(chart_data)
        if image_path and os.path.exists(image_path):
            try:
                doc = FSInputFile(image_path)
                await message.answer_document(
                    doc, caption="🔮 Схема натальной карты"
                )
            except Exception as img_err:
                logger.warning(f"Не удалось отправить SVG: {img_err}")
            finally:
                try:
                    os.remove(image_path)
                except Exception:
                    pass

        # 4. Полный отчёт (шаблон + AI)
        full_report = await natal_service.generate_full_report(chart_data)

        # Отправка по частям (лимит Telegram 4096)
        if len(full_report) > 4000:
            for i in range(0, len(full_report), 4000):
                await message.answer(full_report[i:i + 4000])
        else:
            await message.answer(full_report)

        # 5. Кнопки
        keyboard = InlineKeyboardMarkup(inline_keyboard=[
            [InlineKeyboardButton(text="📄 Скачать PDF-отчёт", callback_data="pdf_natal_from_chart")],
            [InlineKeyboardButton(text="🔙 Главное меню", callback_data="main_menu")],
        ])
        await message.answer("✨ Натальная карта готова!", reply_markup=keyboard)

    except Exception as e:
        logger.error(f"Ошибка создания карты: {e}")
        try:
            await processing_msg.edit_text(f"❌ Ошибка: {str(e)}")
        except Exception:
            await message.answer(f"❌ Ошибка: {str(e)}", reply_markup=back_to_menu_keyboard())

@router.callback_query(F.data == "pdf_natal_from_chart")
async def pdf_natal_from_chart(callback: CallbackQuery):
    user_id = callback.from_user.id
    chart_data = last_chart_cache.get(user_id)

    if not chart_data:
        user_data = await get_user(user_id)
        if not user_data or not user_data.get("birth_date"):
            await callback.message.edit_text(
                "❌ Нужны данные о рождении. Сначала создайте натальную карту.",
                reply_markup=back_to_menu_keyboard()
            )
            return
        try:
            chart_data = await natal_service.create_natal_chart(
                user_data.get("first_name", "Пользователь"),
                user_data["birth_date"],
                user_data.get("birth_time", "12:00"),
                user_data.get("birth_place", "Не указано"),
            )
            last_chart_cache[user_id] = chart_data
        except Exception as e:
            await callback.message.edit_text(
                f"❌ Ошибка: {e}",
                reply_markup=back_to_menu_keyboard()
            )
            return

    processing_msg = await callback.message.edit_text("📄 Создаю PDF...")

    try:
        # AI-интерпретация (если ещё нет)
        if "ai_interpretation" not in chart_data or not chart_data["ai_interpretation"]:
            try:
                ai_text = await natal_service.generate_ai_report(chart_data)
                chart_data["ai_interpretation"] = (ai_text or "")[:1500]
            except Exception as e:
                logger.error(f"Ошибка AI при PDF: {e}")
                chart_data["ai_interpretation"] = ""

        # Создаём PDF
        pdf_path = pdf_gen.create_natal_chart_pdf(chart_data)

        try:
            await processing_msg.delete()
        except Exception:
            pass

        pdf_file = FSInputFile(pdf_path)
        name = chart_data.get("birth_info", {}).get("name", "Пользователь")
        await callback.message.answer_document(
            pdf_file,
            caption=f"🔮 Натальная карта: {name}"
        )
        await callback.message.answer(
            "📌 Выберите действие:",
            reply_markup=main_menu_keyboard(user_id)
        )

        try:
            os.remove(pdf_path)
        except Exception:
            pass

    except Exception as e:
        logger.error(f"Ошибка PDF: {e}")
        try:
            await processing_msg.edit_text(
                f"❌ Ошибка: {str(e)}",
                reply_markup=back_to_menu_keyboard()
            )
        except Exception:
            await callback.message.answer(
                f"❌ Ошибка: {str(e)}",
                reply_markup=back_to_menu_keyboard()
            )

@router.message(TarotStates.waiting_question)
async def tarot_process_question(message: Message, state: FSMContext):
    await state.clear()
    user_id = message.from_user.id
    question = (message.text or "").strip()

    if question == "/cancel":
        await message.answer("Отменено.")
        return

    if len(question) < 5:
        await message.answer("❌ Вопрос слишком короткий. Опишите подробнее.")
        return

    if len(question) > 500:
        question = question[:500]

    # Сообщение о обработке
    processing = await message.answer("🔮 Тасуем карты... Это займёт 5-15 секунд.")

    try:
        result = await get_or_create_reading(user_id, question, count=3)

        if "error" in result:
            await processing.edit_text(f"❌ Ошибка: {result['error']}")
            return

        cards = result["cards"]
        ai_answer = result["ai_answer"]
        from_cache = result.get("from_cache", False)

        try:
            await processing.delete()
        except Exception:
            pass

        # 1. Показываем карты (3 картинки)
        positions = get_positions(question)
        for i, card_key in enumerate(cards):
            img_path = get_card_image(card_key)
            caption = get_card_caption(card_key)
            if i < len(positions):
                caption = f"{positions[i]}: {caption}"

            if img_path and img_path.exists():
                try:
                    await message.answer_photo(
                        FSInputFile(img_path),
                        caption=caption,
                    )
                except Exception as e:
                    logger.warning(f"Не отправить карту {card_key}: {e}")
                    await message.answer(caption)
            else:
                await message.answer(caption)

        # 2. AI-толкование
        cache_note = " (из кэша)" if from_cache else ""
        answer_text = (
            f"🤖 ТОЛКОВАНИЕ АСТРОЛОГА{cache_note}:\n\n"
            f"{ai_answer}"
        )
        if len(answer_text) > 4000:
            for i in range(0, len(answer_text), 4000):
                await message.answer(answer_text[i:i+4000])
        else:
            await message.answer(answer_text)

        # 3. Кнопки
        await message.answer(
            "Хотите ещё один расклад?",
            reply_markup=tarot_back_keyboard(),
        )

    except Exception as e:
        logger.error(f"Ошибка таро: {e}")
        try:
            await processing.edit_text(f"❌ Ошибка: {e}")
        except Exception:
            await message.answer(f"❌ Ошибка: {e}")

def parse_input(text: str) -> dict:
    """
    Возвращает: {
        'type': 'sign' | 'date' | None,
        'value': 'Овен' или '1990-05-15',
        'sign': 'Овен' (всегда если можно определить),
    }
    """
    text_clean = text.strip()

    # По названию знака
    for sign in ZODIAC_SIGNS:
        if sign.lower() in text_clean.lower():
            return {"type": "sign", "value": sign, "sign": sign}

    # По дате — используем универсальный парсер
    from utils.date_parser import parse_date
    ok, iso_date, _ = parse_date(text_clean)
    if ok:
        try:
            date_obj = datetime.strptime(iso_date, "%Y-%m-%d")
            sign = AstrologyCalculator.get_zodiac_sign(date_obj)
            return {"type": "date", "value": iso_date, "sign": sign}
        except Exception:
            pass

    return {"type": None, "value": None, "sign": None}

def build_cache_key(p1: dict, p2: dict) -> str:
    """Строит ключ кэша по введённым данным."""
    if p1["type"] == "date" and p2["type"] == "date":
        # Сортируем даты
        dates = sorted([p1["value"], p2["value"]])
        return f"compat_{dates[0]}_{dates[1]}"
    else:
        # По знакам (сортируем)
        signs = sorted([p1["sign"].lower(), p2["sign"].lower()])
        return f"compat_{signs[0]}_{signs[1]}"

def fallback_compatibility(sign1: str, sign2: str) -> dict:
    """Локальный расчёт совместимости."""
    key = (sign1, sign2)
    score = BASE_SCORES.get(key, BASE_SCORES.get((sign2, sign1), random.randint(55, 90)))

    if score >= 90:
        level = "Идеальная совместимость"
    elif score >= 80:
        level = "Отличная совместимость"
    elif score >= 70:
        level = "Хорошая совместимость"
    elif score >= 60:
        level = "Средняя совместимость"
    else:
        level = "Сложная совместимость"

    element1 = ELEMENTS.get(sign1, "?")
    element2 = ELEMENTS.get(sign2, "?")
    planet1 = RULING_PLANETS.get(sign1, "?")
    planet2 = RULING_PLANETS.get(sign2, "?")
    quality1 = QUALITIES.get(sign1, "?")
    quality2 = QUALITIES.get(sign2, "?")

    text = (
        f"💕 АНАЛИЗ СОВМЕСТИМОСТИ\n\n"
        f"✨ АСТРОЛОГИЧЕСКИЕ ДАННЫЕ:\n"
        f"• Стихии: {element1} и {element2}\n"
        f"• Планеты: {planet1} и {planet2}\n"
        f"• Качества: {quality1} и {quality2}\n\n"
        f"🌊 ГАРМОНИЯ СТИХИЙ:\n"
        f"Сочетание {element1} и {element2} создаёт уникальную динамику.\n\n"
        f"💫 РЕКОМЕНДАЦИЯ:\n"
        f"Учитесь слушать друг друга. Ваш союз — это сила!\n\n"
        f"🔮 Помните: астрология — это руководство, а не приговор."
    )

    return {"score": score, "level": level, "text": text}

@router.message(CompatibilityStates.waiting_partner2)
async def process_partner2(message: Message, state: FSMContext):
    if message.text == "/cancel":
        await state.clear()
        await message.answer("Отменено", reply_markup=back_to_menu_keyboard())
        return

    p2 = parse_input(message.text)
    if not p2["sign"]:
        await message.answer(
            "❌ Не удалось определить знак. Попробуйте ещё раз."
        )
        return

    data = await state.get_data()
    p1 = data["p1"]
    await state.clear()

    processing_msg = await message.answer("🔮 Астролог анализирует совместимость...")

    # Ключ кэша
    cache_key = build_cache_key(p1, p2)

    # Проверка кэша
    result = None
    from_cache = False

    try:
        cached = await get_setting(cache_key)
        if cached and isinstance(cached, dict) and cached.get("text"):
            result = cached
            from_cache = True
            logger.info(f"✅ Совместимость из кэша: {cache_key}")
    except Exception as e:
        logger.warning(f"Ошибка кэша совместимости: {e}")

    # Запрос к AI
    if not result:
        try:
            ai_result = await ai_service.generate_compatibility(p1["sign"], p2["sign"])
            if ai_result:
                result = ai_result
                await set_setting(cache_key, ai_result)
                logger.info(f"✅ Новая совместимость сохранена: {cache_key}")
        except Exception as e:
            logger.error(f"Ошибка AI-совместимости: {e}")

    # Fallback
    if not result:
        logger.warning("⚠️ Fallback совместимости")
        result = fallback_compatibility(p1["sign"], p2["sign"])

    # Форматируем вывод
    score = result.get("score", 75)
    level = result.get("level", "Совместимость")
    text = result.get("text", "")

    stars = "⭐" * (score // 10) + "☆" * (10 - (score // 10))

    final_text = (
        f"💕 АНАЛИЗ СОВМЕСТИМОСТИ\n\n"
        f"{p1['sign']} 💞 {p2['sign']}\n\n"
        f"{stars}\n"
        f"{score}% — {level}\n\n"
        f"━━━━━━━━━━━━━━━━━━━━━━━━\n\n"
        f"{text}"
    )

    try:
        await processing_msg.delete()
    except Exception:
        pass

    await message.answer(final_text, reply_markup=main_menu_keyboard(message.from_user.id))

@router.message(AIQuestionStates.waiting_question)
async def process_ai_question(message: Message, state: FSMContext):
    user_id = message.from_user.id

    # Отмена
    if message.text == "/cancel":
        await state.clear()
        await message.answer(
            "Диалог завершён.",
            reply_markup=back_to_menu_keyboard()
        )
        return

    # Валидация
    valid, result = validator.validate_question(message.text)
    if not valid:
        await message.answer(
            f"❌ {result}\n\nПожалуйста, задайте осмысленный вопрос.",
            reply_markup=ai_dialog_keyboard()
        )
        return

    # Проверка лимита
    limit = await get_user_ai_limit(user_id)
    usage = await get_ai_usage(user_id)

    if limit != -1 and usage >= limit:
        await state.clear()
        keyboard = InlineKeyboardMarkup(inline_keyboard=[
            [InlineKeyboardButton(text="💎 Купить Премиум", callback_data="check_status")],
            [InlineKeyboardButton(text="🔙 Главное меню", callback_data="main_menu")],
        ])
        await message.answer(
            "⚠️ ЛИМИТ ВОПРОСОВ ИСЧЕРПАН\n\n"
            f"Вы задали {usage}/{limit} вопросов сегодня.\n"
            "Лимит обновится в 00:00 UTC.",
            reply_markup=keyboard
        )
        return

    # Сообщение о обработке
    processing_msg = await message.answer("🔮 Астролог анализирует ваш вопрос...")

    async def update_thinking():
        await asyncio.sleep(3)
        try:
            await processing_msg.edit_text(
                "🔮 Астролог думает...\n\n"
                "Это может занять 10-30 секунд.\n"
                "Пожалуйста, подождите."
            )
        except Exception:
            pass

    thinking_task = asyncio.create_task(update_thinking())

    try:
        response = await ai_service.process_question(user_id, result)
        thinking_task.cancel()

        try:
            await processing_msg.delete()
        except Exception:
            pass

        answer = response.get("message", "Ошибка ответа")
        from_cache = response.get("from_cache", False)
        facts = response.get("facts", [])

        if not from_cache:
            await increment_ai_usage(user_id)

        admin_marker = ""
        if await is_admin(user_id):
            admin_marker = f"\n\n🤖 Модель: {response.get('model', '?')}"

        # Ответ + кнопка «Закончить диалог»
        await message.answer(
            f"🤖 ОТВЕТ АСТРОЛОГА:\n\n{answer}{admin_marker}\n\n"
            f"💬 Можете задать следующий вопрос или нажать «🔚 Закончить диалог».",
            reply_markup=ai_dialog_keyboard()
        )

        # Остаток лимита
        if limit != -1:
            new_usage = await get_ai_usage(user_id)
            left = max(0, limit - new_usage)
            if left <= 5 and left > 0:
                await message.answer(f"⚠️ У вас осталось {left} запросов на сегодня.")
            elif left == 0:
                await message.answer("⚠️ Лимит вопросов исчерпан на сегодня.")

        # Факты
        if facts:
            existing_notes = await get_user_notes(user_id)
            existing_texts = [n["note"] for n in existing_notes]
            new_facts = [f for f in facts if f not in existing_texts]

            if new_facts:
                await show_facts_confirmation(message, state, new_facts)
            else:
                await message.answer(
                    "✅ Астролог заметил, что вы уже рассказывали об этом ранее."
                )
        else:
            q_lower = message.text.lower()
            personal_pronouns = ["я ", "мне ", "меня ", "мой ", "моя ", "мои "]
            if any(p in q_lower for p in personal_pronouns):
                keyboard = InlineKeyboardMarkup(inline_keyboard=[
                    [InlineKeyboardButton(text="👤 Заполнить профиль", callback_data="profile_edit")],
                    [InlineKeyboardButton(text="🔚 Закончить диалог", callback_data="ai_end_dialog")],
                ])
                await message.answer(
                    "💡 Хотите, чтобы астролог знал о вас больше?\n\n"
                    "Заполните профиль — и ответы станут точнее.",
                    reply_markup=keyboard
                )

        # ВАЖНО: FSM НЕ сбрасывается! Пользователь может продолжить.

    except Exception as e:
        thinking_task.cancel()
        logger.error(f"Ошибка AI: {e}")
        try:
            await processing_msg.edit_text(
                "❌ Произошла ошибка. Попробуйте позже.",
                reply_markup=ai_dialog_keyboard()
            )
        except Exception:
            await message.answer(
                "❌ Произошла ошибка.",
                reply_markup=ai_dialog_keyboard()
            )

async def show_facts_confirmation(message: Message, state: FSMContext, facts: list):
    # Сохраняем факты в отдельном ключе FSM (не сбрасывая состояние диалога)
    current_data = await state.get_data()
    current_data["pending_facts"] = facts
    current_data["pending_selected"] = list(range(len(facts)))
    await state.update_data(**current_data)

    keyboard = _build_facts_keyboard(facts, list(range(len(facts))))
    await message.answer(
        "✨ АСТРОЛОГ ХОЧЕТ ЗАПОМНИТЬ О ВАС\n\n"
        "Нажмите на факт, чтобы выбрать/убрать.\n"
        "Затем нажмите «💾 Сохранить».",
        reply_markup=keyboard
    )


def _build_facts_keyboard(facts: list, selected: list):
    buttons = []
    for i, fact in enumerate(facts):
        mark = "✅" if i in selected else "☐"
        short = fact[:40] + "..." if len(fact) > 40 else fact
        buttons.append([InlineKeyboardButton(
            text=f"{mark} {short}",
            callback_data=f"fact_toggle_{i}"
        )])
    buttons.append([
        InlineKeyboardButton(text="💾 Сохранить", callback_data="facts_save"),
        InlineKeyboardButton(text="❌ Не надо", callback_data="facts_cancel"),
    ])
    return InlineKeyboardMarkup(inline_keyboard=buttons)


@router.callback_query(F.data.startswith("fact_toggle_"))
async def fact_toggle(callback: CallbackQuery, state: FSMContext):
    data = await state.get_data()
    facts = data.get("pending_facts", [])
    selected = data.get("pending_selected", [])
    idx = int(callback.data.replace("fact_toggle_", ""))

    if idx in selected:
        selected.remove(idx)
    else:
        selected.append(idx)

    await state.update_data(pending_selected=selected)
    try:
        await callback.message.edit_reply_markup(
            reply_markup=_build_facts_keyboard(facts, selected)
        )
    except Exception:
        pass
    await callback.answer()

async def mood_story(message: Message, state: FSMContext, bot: Bot):

    mood = data.get("current_mood", 3)
    streak = data.get("streak", 0)
    user_id = message.from_user.id

    # /cancel
    if message.text == "/cancel":

        await message.answer("Хорошо. До встречи!", reply_markup=main_menu_keyboard(user_id))
        return

    text = message.text.strip()
    if len(text) < 3:
        await message.answer("Расскажите чуть подробнее или нажмите «⏭️ Пропустить».")
        return

    if len(text) > 2000:
        text = text[:2000]

    # Проверка лимита
    limit = await get_user_ai_limit(user_id)
    usage = await get_ai_usage(user_id)
    if limit != -1 and usage >= limit:
        await state.clear()
        keyboard = InlineKeyboardMarkup(inline_keyboard=[
            [InlineKeyboardButton(text="💎 Купить Премиум", callback_data="check_status")],
            [InlineKeyboardButton(text="🔙 Главное меню", callback_data="main_menu")],
        ])
        await message.answer(
            "⚠️ ЛИМИТ ВОПРОСОВ ИСЧЕРПАН\n\n"
            f"Вы задали {usage}/{limit} вопросов сегодня.\n"
            "Лимит обновится в 00:00 UTC.\n\n"
            "💎 Купите Премиум — 100 запросов/день.\n\n"
            "Ваше настроение сохранено 🌟",
            reply_markup=keyboard,
        )
        return

    # Отправляем «AI думает»
    processing = await message.answer("🔮 Астролог обрабатывает ваш рассказ...")

    try:
        # 1. AI-ответ (совет) на основе настроения + текста
        from ai_service_n1n import ai_service

        mood_text = f"{mood_emoji(mood)} {mood_name(mood)}"
        prompt = (
            f"Пользователь отметил настроение: {mood_text} (шкала: 5=отлично, 1=ужасно).\n"
            f"Рассказ пользователя: {text}\n\n"
            f"Ответь от лица заботливого астролога:\n"
            f"- Поддержи или порадуйся\n"
            f"- Дай 1-2 коротких совета на основе настроения\n"
            f"- Упомяни астрологию, если уместно\n"
            f"- Без markdown (**, __, `)\n"
            f"- 3-5 предложений\n"
        )

        ai_response = None
        try:
            ai_response = await ai_service._query_ai(
                "Ты — заботливый астролог. Отвечай тёпло, поддерживающе, до 5 предложений. Без markdown.",
                prompt,
                max_tokens=500,
            )
            if ai_response:
                ai_response = ai_service._clean_response(ai_response)
        except Exception as e:
            logger.error(f"Ошибка AI mood: {e}")

        if not ai_response:
            # Fallback
            if mood >= 4:
                ai_response = "Рад за вас! Пусть этот позитив заряжает вас на всю неделю ✨"
            elif mood <= 2:
                ai_response = "Держитесь! Помните: даже самая тёмная ночь заканчивается рассветом 🌅"
            else:
                ai_response = "Спасибо, что поделились. Желаю вам гармонии и спокойствия ✨"

        try:
            await processing.delete()
        except Exception:
            pass

        # 2. Извлекаем факты (через AI)
        facts = []
        try:
            facts = await ai_service.extract_user_facts(text, ai_response or "")
        except Exception as e:
            logger.debug(f"Ошибка извлечения фактов mood: {e}")

        # 3. Тратим лимит ТОЛЬКО при успешном AI-ответе
        if ai_response:
            try:
                await increment_ai_usage(user_id)
            except Exception:
                pass

        # 4. Отправляем AI-ответ
        await message.answer(f"🤖 {ai_response}")

        # 4. Предлагаем сохранить факты (если есть)
        if facts:
            existing_notes = await get_user_notes(user_id)
            existing_texts = [n["note"] for n in existing_notes]
            new_facts = [f for f in facts if f not in existing_texts]

            if new_facts:
                # Сохраняем в state для подтверждения
                await state.update_data(pending_facts=new_facts, pending_selected=list(range(len(new_facts))))
                keyboard = _build_facts_keyboard(new_facts, list(range(len(new_facts))))
                await message.answer(
                    "✨ Астролог хочет запомнить о вас:\n\n"
                    "Нажмите на факт, чтобы выбрать/убрать.\n"
                    "Затем нажмите «💾 Сохранить».",
                    reply_markup=keyboard,
                )
                return  # FSM не сбрасываем — ждём подтверждения

        # 5. Если фактов нет — выходим из FSM
        await state.clear()
        await message.answer(
            "Хорошего дня! 🌟",
            reply_markup=main_menu_keyboard(user_id),
        )

    except Exception as e:
        logger.error(f"Ошибка mood_story: {e}")
        try:
            await processing.edit_text("❌ Произошла ошибка. Попробуйте ещё раз.")
        except Exception:
            pass
        await state.clear()

def _build_facts_keyboard(facts: list, selected: list):
    buttons = []
    for i, fact in enumerate(facts):
        mark = "✅" if i in selected else "☐"
        short = fact[:40] + "..." if len(fact) > 40 else fact
        buttons.append([InlineKeyboardButton(
            text=f"{mark} {short}",
            callback_data=f"mood_fact_toggle_{i}",
        )])
    buttons.append([
        InlineKeyboardButton(text="💾 Сохранить", callback_data="mood_facts_save"),
        InlineKeyboardButton(text="❌ Не надо", callback_data="mood_facts_cancel"),
    ])
    return InlineKeyboardMarkup(inline_keyboard=buttons)

    )
    await callback.message.answer(
        "Хорошего дня! 🌟",
        reply_markup=main_menu_keyboard(callback.from_user.id),
    )

@router.callback_query(F.data == "mood_chart")
async def mood_chart(callback: CallbackQuery):
    user_id = callback.from_user.id
    history = await get_mood_history(user_id, days=30)

    if len(history) < 2:
        await callback.answer("Нужно минимум 2 записи для графика.", show_alert=True)
        return

    msg = await callback.message.edit_text("📈 Строю график...")
    try:
        chart_path = _build_mood_chart(history)
        if not chart_path:
            await msg.edit_text("❌ Не удалось построить график.")
            return
        try:
            await msg.delete()
        except Exception:
            pass
        photo = FSInputFile(chart_path)
        await callback.message.answer_photo(photo, caption="📈 Настроение за 30 дней")
        await callback.message.answer(
            "Что дальше?",
            reply_markup=InlineKeyboardMarkup(inline_keyboard=[
                [InlineKeyboardButton(text="📔 Дневник", callback_data="mood_diary")],
                [InlineKeyboardButton(text="📊 Статистика", callback_data="mood_stats")],
                [InlineKeyboardButton(text="🔙 Главное меню", callback_data="main_menu")],
            ]),
        )
        try:
            os.remove(chart_path)
        except Exception:
            pass
    except Exception as e:
        logger.error(f"Ошибка графика: {e}")
        try:
            await msg.edit_text(f"❌ Ошибка: {e}")
        except Exception:
            pass


def _build_mood_chart(history: list) -> str:
    try:
        import matplotlib
        matplotlib.use("Agg")
        import matplotlib.pyplot as plt
    except ImportError:
        logger.error("matplotlib не установлен")
        return None

    data = sorted(history, key=lambda x: x["mood_date"])
    dates = [d["mood_date"].strftime("%d.%m") for d in data]
    moods = [d["mood"] for d in data]

    fig, ax = plt.subplots(figsize=(10, 5))
    fig.patch.set_facecolor("#1a2a3a")
    ax.set_facecolor("#1a2a3a")
    ax.plot(dates, moods, "o-", color="#FFD700", linewidth=2, markersize=8)
    ax.grid(True, alpha=0.3, color="#c9a959")
    ax.set_ylim(0.5, 5.5)
    ax.set_yticks([1, 2, 3, 4, 5])
    ax.set_yticklabels(["😢", "😔", "😐", "🙂", "😊"], fontsize=14)
    ax.set_title("Настроение за 30 дней", color="#FFD700", fontsize=16, pad=15)
    ax.tick_params(colors="#F0F0F0", labelsize=9)
    plt.xticks(rotation=45, ha="right")
    for spine in ax.spines.values():
        spine.set_color("#c9a959")
    plt.tight_layout()
    filename = f"mood_chart_{uuid.uuid4().hex}.png"
    plt.savefig(filename, facecolor="#1a2a3a", dpi=100)
    plt.close()
    return filename

@router.callback_query(F.data == "daily_horoscope")
async def daily_horoscope_handler(callback: CallbackQuery):
    """Гороскоп дня — AI + кэш."""
    today_str = datetime.now().strftime("%d.%m.%Y")

    # Проверяем кэш
    cached = await get_daily_horoscope()
    if cached:
        logger.info("✅ Гороскоп из кэша")
        try:
            await callback.message.edit_text(
                cached, reply_markup=back_to_menu_keyboard()
            )
        except Exception:
            await callback.message.answer(
                cached, reply_markup=back_to_menu_keyboard()
            )
        return

    # Генерируем новый
    processing_msg = None
    try:
        processing_msg = await callback.message.edit_text(
            "🌅 Генерирую гороскоп на сегодня...\n\n"
            "Это займёт 5-15 секунд."
        )
    except Exception:
        processing_msg = await callback.message.answer(
            "🌅 Генерирую гороскоп на сегодня..."
        )

    # Запрос к AI
    try:
        text = await ai_service.generate_horoscope()
    except Exception as e:
        logger.error(f"Ошибка AI-гороскопа: {e}")
        text = None

    if text:
        # Сохраняем в кэш
        await set_daily_horoscope(text)
        logger.info(f"✅ Новый AI-гороскоп сохранён: {len(text)} символов")
    else:
        # Fallback
        logger.warning("⚠️ AI не ответил, используем fallback")
        text = FALLBACK_HOROSCOPE.format(date=today_str)

    # Отправляем
    try:
        await processing_msg.edit_text(text, reply_markup=back_to_menu_keyboard())
    except Exception:
        await callback.message.answer(text, reply_markup=back_to_menu_keyboard())
